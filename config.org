(defvar bootstrap-version)
(let ((bootstrap-file
       (expand-file-name
        "straight/repos/straight.el/bootstrap.el"
        (or (bound-and-true-p straight-base-dir)
            user-emacs-directory)))
      (bootstrap-version 7))
  (unless (file-exists-p bootstrap-file)

        (url-retrieve-synchronously
         "https://raw.githubusercontent.com/radian-software/straight.el/develop/install.el"
         'silent 'inhibit-cookies)
      (goto-char (point-max))
      (eval-print-last-sexp)))
  (load bootstrap-file nil 'nomessage))

(straight-use-package 'use-package)

(setq straight-use-package-by-default t)

** Reproducible package versions
Package versions are *pinned* in =straight/versions/default.el= (created with
=M-x straight-freeze-versions=), so the same commits are checked out on every
machine and startup.  This replaces =auto-package-update= — a =package.el= tool
that does not cooperate with straight.el and silently pulls breaking upstream
changes.

To update deliberately:
- =M-x straight-pull-all= — fetch/checkout the latest for every package
- =M-x straight-freeze-versions= — re-pin the lockfile to the new commits

  "Automatically add newline and indent between braces."
  (when (and (eq (char-before) ?\{)

    (newline)
    (indent-according-to-mode)

* Competitive Programming
A local "compile-and-run" workflow. Drop an =input.txt= next to your source and
it is fed to the program on stdin; =output.txt= mirrors stdout live. See the
function-key bindings in *Competitive Programming Keybindings* below.
#+begin_src emacs-lisp

(defvar +cp-cxx-standard "c++20"
  "C++ standard used by the competitive-programming runner.")
(defvar +cp-c-standard "c17"
  "C standard used by the competitive-programming runner.")
(defvar +cp-build-flags "-Wall -Wextra -Wshadow -O2 -DLOCAL"
  "Extra compiler flags for the competitive-programming runner.")
(defvar +cp-cxx-include-dir (expand-file-name "~/.config/cpp/include")
  "Include dir holding a libc++-compatible <bits/stdc++.h> for C++ builds.
macOS Clang/libc++ lacks GCC's <bits/stdc++.h>; this dir provides it.")
(defvar +cp-write-output-file t
  "When non-nil, tee the program's stdout into `output.txt' beside the source.
This is what keeps the bottom-right pane of `+cp/layout' live after every run.")

;; The starter skeleton used by `+cp/new-problem'. One entry per language; the
;; right one is chosen from the current buffer's major mode (C++ is the default).
(defvar +cp-templates
  `((c++-mode . ,(concat "#include <bits/stdc++.h>\n"
                         "using namespace std;\n"
                         "\n"
                         "#define all(x) (x).begin(), (x).end()\n"
                         "#define sz(x) (int)(x).size()\n"
                         "using ll = long long;\n"
                         "using pii = pair<int,int>;\n"
                         "\n"
                         "void solve() {\n"
                         "    \n"
                         "}\n"
                         "\n"
                         "int main() {\n"
                         "    ios::sync_with_stdio(false);\n"
                         "    cin.tie(nullptr);\n"
                         "    int t = 1;\n"
                         "    // cin >> t;\n"
                         "    while (t--) solve();\n"
                         "    return 0;\n"
                         "}\n"))
    (python-mode . ,(concat "import sys\n"
                            "input = sys.stdin.readline\n"
                            "\n"
                            "def solve():\n"
                            "    pass\n"
                            "\n"
                            "def main():\n"
                            "    t = 1\n"
                            "    # t = int(input())\n"
                            "    for _ in range(t):\n"
                            "        solve()\n"
                            "\n"
                            "if __name__ == \"__main__\":\n"
                            "    main()\n")))
  "Alist of (MAJOR-MODE . TEMPLATE-STRING) for `+cp/new-problem'.")

(defun +cp--stdin-suffix (dir)
  "Return a ` < input.txt' shell suffix if that file exists in DIR, else \"\"."
  (let ((in (expand-file-name "input.txt" dir)))
    (if (file-exists-p in)
        (format " < %s" (shell-quote-argument in))
      "")))

(defun +cp/compile-and-run ()
  "Build and run the current file. Acts as a universal F5 \"build\" key.

In LaTeX buffers it runs the full latexmk build. In C/C++/Python/Java buffers
it compiles (when needed) and runs the program, piping `input.txt' on stdin
when that file is present beside the source."
  (interactive)
  (when buffer-file-name (save-buffer))
  (pcase major-mode
    ((or 'latex-mode 'LaTeX-mode)
     (TeX-command-run-all nil))
    ((guard (null buffer-file-name))
     (user-error "Buffer is not visiting a file"))
    (_
     (let* ((file  (file-name-nondirectory buffer-file-name))
            (base  (file-name-sans-extension file))
            (dir   (file-name-directory buffer-file-name))
            (stdin (+cp--stdin-suffix dir))
            (qbase (shell-quote-argument base))
            (qfile (shell-quote-argument file))
            (cmd
             (pcase major-mode
               ((or 'c++-mode 'c++-ts-mode)
                (format "g++ -std=%s %s -I%s -o %s %s && ./%s%s"
                        +cp-cxx-standard +cp-build-flags
                        (shell-quote-argument +cp-cxx-include-dir)
                        qbase qfile qbase stdin))
               ((or 'c-mode 'c-ts-mode)
                (format "gcc -std=%s %s -o %s %s && ./%s%s"
                        +cp-c-standard +cp-build-flags qbase qfile qbase stdin))
               ((or 'python-mode 'python-ts-mode)
                (format "python3 %s%s" qfile stdin))
               ((or 'java-mode 'java-ts-mode)
                (format "javac %s && java %s%s" qfile qbase stdin))
               (_ (user-error "No competitive runner for %s" major-mode)))))
       ;; Mirror the program's stdout into `output.txt' (the layout's bottom-right
       ;; pane) while still showing it in the *compilation* buffer. `&&' binds
       ;; tighter than `|', so tee only ever wraps the final run command.
       (when +cp-write-output-file
         (setq cmd (format "%s | tee %s" cmd
                           (shell-quote-argument
                            (expand-file-name "output.txt" dir)))))
       ;; Force the run log into the layout's output pane (overriding popups)
       ;; so it shows there, then flips back to output.txt on success.
       (let* ((default-directory dir)
              (compilation-scroll-output t)
              (owin (+cp--output-window))
              (display-buffer-overriding-action
               (if owin
                   (list (lambda (buf _alist)
                           (with-current-buffer buf
                             (setq-local +cp--in-output-window t))
                           (set-window-buffer owin buf)
                           owin))
                 display-buffer-overriding-action)))
         (compile cmd))))))

(defun +cp/edit-input ()
  "Open (creating if needed) `input.txt' beside the current file in a side window."
  (interactive)
  (unless buffer-file-name (user-error "Buffer is not visiting a file"))
  (let ((in (expand-file-name "input.txt" (file-name-directory buffer-file-name))))
    (find-file-other-window in)))

(defun +cp--touch (file)
  "Create FILE on disk if it does not exist yet, and return FILE."
  (unless (file-exists-p file)
    (write-region "" nil file nil 'silent))
  file)

(defun +cp--truncate (file)
  "Empty FILE on disk and refresh any buffer visiting it."
  (write-region "" nil file nil 'silent)
  (when-let ((buf (find-buffer-visiting file)))
    (with-current-buffer buf (revert-buffer t t t))))

(defun +cp/layout ()
  "Lay out the competitive-programming IDE: source left, I/O stacked right.

The current source file fills the left half; `input.txt' (top) and `output.txt'
\(bottom) sit in the right half. Both side files are created if missing, and the
output pane auto-reverts so each run's result shows up immediately."
  (interactive)
  (unless buffer-file-name (user-error "Buffer is not visiting a file"))
  (let* ((dir    (file-name-directory buffer-file-name))
         (input  (+cp--touch (expand-file-name "input.txt"  dir)))
         (output (+cp--touch (expand-file-name "output.txt" dir)))
         (src    (current-buffer)))
    (delete-other-windows)
    (let ((right (split-window-right)))
      (with-selected-window right
        (find-file input)
        (with-selected-window (split-window-below)
          (find-file output)
          (auto-revert-mode 1))))   ; live-refresh output.txt after runs
    ;; Land back on the source, sized to the golden ratio by zoom-mode.
    (select-window (get-buffer-window src))))

(defun +cp/new-problem ()
  "Reset the buffer to its language template and clear `input.txt'/`output.txt'.
One keypress to start a fresh problem. Asks for confirmation first."
  (interactive)
  (unless buffer-file-name (user-error "Buffer is not visiting a file"))
  (when (yes-or-no-p "New problem — reset this file and clear input/output? ")
    (let* ((dir  (file-name-directory buffer-file-name))
           (mode (pcase major-mode
                   ((or 'c++-mode 'c++-ts-mode 'c-mode 'c-ts-mode) 'c++-mode)
                   ((or 'python-mode 'python-ts-mode) 'python-mode)
                   (m m)))
           (tmpl (alist-get mode +cp-templates)))
      (unless tmpl (user-error "No template for %s" major-mode))
      (erase-buffer)
      (insert tmpl)
      (save-buffer)
      (+cp--truncate (expand-file-name "input.txt"  dir))
      (+cp--truncate (expand-file-name "output.txt" dir))
      ;; Drop the cursor inside solve() (or its first blank body line).
      (goto-char (point-min))
      (when (re-search-forward "solve()\\s-*[:{]\n" nil t)
        (end-of-line)
        (when (memq mode '(c++-mode)) (indent-for-tab-command)))
      (message "Fresh problem ready."))))

(defun +cp--revert-output-buffers (&rest _)
  "Refresh any visible `output.txt' buffer after a compile finishes."
  (dolist (buf (buffer-list))
    (let ((f (buffer-file-name buf)))
      (when (and f (string= (file-name-nondirectory f) "output.txt")
                 (file-exists-p f))
        (with-current-buffer buf (revert-buffer t t t))))))
(add-hook 'compilation-finish-functions #'+cp--revert-output-buffers)

;; Run log that lives in the output pane. When you build, the bottom-right pane
;; (showing `output.txt') is reused to display the *compilation* log. A few
;; seconds after a CLEAN build the pane flips back to the saved file; on a
;; compile ERROR the log stays put so you can read the diagnostics.
(defvar +cp-compilation-popup-timeout 3
  "Seconds the run log stays in the output pane before flipping back to the file.
Set to nil to keep the log up until your next build.")

(defvar-local +cp--in-output-window nil
  "Non-nil in a *compilation* buffer shown inside the layout's output pane.")

(defun +cp--output-window ()
  "Return the window currently showing an `output.txt' file, if any."
  (seq-find (lambda (w)
              (let ((f (buffer-file-name (window-buffer w))))
                (and f (string= (file-name-nondirectory f) "output.txt"))))
            (window-list)))

(defun +cp--restore-output-window (buf status)
  "Flip the output pane from the run log BUF back to `output.txt' after success."
  (when (and +cp-compilation-popup-timeout
             (bufferp buf)
             (buffer-local-value '+cp--in-output-window buf)
             (stringp status)
             (string-match-p "finished" status))
    (let ((dir (buffer-local-value 'default-directory buf)))
      (run-at-time
       +cp-compilation-popup-timeout nil
       (lambda ()
         (when-let ((win (get-buffer-window buf)))
           (let* ((out  (expand-file-name "output.txt" dir))
                  (obuf (or (find-buffer-visiting out)
                            (and (file-exists-p out) (find-file-noselect out)))))
             (when obuf
               (with-current-buffer obuf
                 (when (file-exists-p out) (revert-buffer t t t)))
               (set-window-buffer win obuf)))))))))
(add-hook 'compilation-finish-functions #'+cp--restore-output-window)

#+end_src
